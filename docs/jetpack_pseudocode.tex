\title{Jetpack Algorithm (Non-Optimized Reference)}
\maketitle

\paragraph{Overview}
This document presents the conceptual (non-optimized) Jetpack algorithm. It is the direct formalization of the Jetpack protocol logic without the implementation optimizations described in \texttt{jetpack\_pseudocode\_optimized.tex}. The normal fast path is identical to the optimized version. The recovery algorithm uses explicit per-key Prepare/Accept/Commit rounds, making the protocol structure clearer at the cost of additional round trips.

\paragraph{Relationship to Optimized Version}
The non-optimized recovery algorithm performs:
\begin{enumerate}
  \setlength{\itemsep}{0pt}
  \item One \texttt{BeginRecovery} broadcast (to propagate new view and collect all witness key-ids at once).
  \item $N$ sequential per-key \texttt{Prepare} broadcasts — one for each recovered key $k$ in the collected pool.
  \item $N$ per-key \texttt{Accept} broadcasts (one per key, after each \texttt{Prepare} succeeds).
  \item $N$ per-key \texttt{Commit} broadcasts (one per key, after each \texttt{Accept} succeeds).
  \item One \texttt{Resubmit} + \texttt{FinishRecovery} sequence.
\end{enumerate}
Total round trips: $1 + N + N + $ fire-and-forget, where $N$ is the number of in-flight fast-path commands at the time of failure. The optimized version (see \texttt{jetpack\_pseudocode\_optimized.tex}) reduces this to 2 constant-latency rounds regardless of $N$.

\paragraph{State Variables}
Each replica $R$ maintains:
$R.jepoch$ (Jetpack epoch), $R.oepoch$ (original protocol epoch),
$R.old\_view$ / $R.new\_view$ (views from original protocol),
$R.status \in \{RDY, REC\}$,
$R.witness$ (per-key candidate set: $candidates[key]$, $seen\_ballot[key]$, $acc\_ballot[key]$, $value[key]$, $committed[key]$),
$R.rec\_set$ (map $sid \to \{(key, cmd)\}$).


%% ============================================================
%%  NORMAL PATH (same as optimized)
%% ============================================================

\begin{algorithm}
\caption{Client -- Dispatch (fast path)}
\label{alg:client_dispatch}
\setcounter{AlgoLine}{0}

\underline{Client $C$, dispatching $cmd$ on fast path}\\
$(cmd\_id, key, value) \gets cmd$\\
broadcast $Dispatch(C.jepoch, cmd)$ to all replicas in $C.view$\\
\eIf{receive $\ge f+1$ $DispatchAck(accepted{=}true, result)$
  \textbf{and} all proposing replicas replied $accepted{=}true$\label{line:fastpathcond}}{
    return $result$ to application\\
}{
    wait for $result$ of $cmd\_id$ from the original protocol path\\
    return $result$ to application\\
}
\If{receive $DispatchAck(accepted{=}false, jepoch', view')$
    \textbf{with} $jepoch' > C.jepoch$}{
    $C.jepoch \gets jepoch'$; \; $C.view \gets view'$\\
}
\end{algorithm}


\begin{algorithm}
\caption{Server -- Witness Conflict Check (on $Dispatch$)}
\label{alg:witness_check}
\setcounter{AlgoLine}{10}

\underline{Replica $R$, on receiving $Dispatch(jepoch, cmd)$ from client}\\
\If{$jepoch \ne R.jepoch$ \textbf{or} $R.status = REC$}{
    reply $DispatchAck(accepted{=}false, R.jepoch, R.new\_view)$\\
    \KwRet\\
}
$(key, cmd\_id) \gets$ parse $cmd$\\
\eIf{$R.witness.candidates[key]$ has no conflicting write}{
    $R.witness.candidates[key].insert(cmd\_id, cmd)$\\
    \If{$R$ is proposing replica}{
        $result \gets speculative\_install(cmd)$\\
    }
    reply $DispatchAck(accepted{=}true, result)$\\
}{
    $R.witness.candidates[key].insert(cmd\_id, cmd)$\\
    reply $DispatchAck(accepted{=}false, R.jepoch, R.new\_view)$\\
}
\If{$R$ is proposing replica}{
    $OnRequest(cmd)$\tcp*{submit to original protocol}
}
\end{algorithm}


\begin{algorithm}
\caption{Original Protocol -- Execute (with GC)}
\label{alg:execute_gc}
\setcounter{AlgoLine}{26}

\underline{Replica $R$, ready to execute $cmd$}\\
\If{$cmd \notin R.executed$}{
    $OriginalProtocolExecute(cmd)$\\
    $R.executed.insert(cmd)$\tcp*{deduplication}
}
$R.witness.remove(cmd)$\tcp*{garbage collection}
\end{algorithm}

\paragraph{Original Protocol Hook}
When the original protocol completes a view change (elects a new proposer), it broadcasts \texttt{BeginRecovery} via Jetpack \emph{before} accepting new requests. \texttt{BeginRecovery} both propagates the new view and collects witness candidates in a single round. The coordinator then runs one Prepare/Accept/Commit cycle \emph{per recovered key} sequentially.


\begin{algorithm}
\caption{Original -- Trigger Recovery}
\label{alg:trigger_recovery}
\setcounter{AlgoLine}{32}

\underline{Replica $R$ completes view change from $old\_view$ to $new\_view$}\\
\tcc{Called by original protocol hook: pause before accepting new reqs}
broadcast $BeginRecovery(old\_view, new\_view, R.oepoch)$ to all replicas in $old\_view$\label{line:broadcastbegin}\\
wait for $f+1$ $BeginRecoveryAck$\label{line:waitbegin}\\
$R.Recovery()$\tcp*{run per-key recovery rounds}
\end{algorithm}

\paragraph{View Change Hook}
The Jetpack hook is inserted between the moment the new proposer is elected and the moment it begins accepting requests (line~\ref{line:broadcastbegin}). Since different protocols place this boundary differently (Raft: after election, before serving; CoPilot: after leader bump; MongoDB: after primary step-up), the hook is inserted at the protocol-specific transition point. Line~\ref{line:waitbegin} ensures a quorum of replicas is in recovery mode before the coordinator begins collecting commands.


\begin{algorithm}
\caption{Server -- $OnBeginRecovery$}
\label{alg:on_begin_recovery}
\setcounter{AlgoLine}{37}

\underline{Replica $R$, on receiving $BeginRecovery(old\_view, new\_view, oepoch)$}\\
$R.old\_view \gets old\_view$\\
$R.new\_view \gets new\_view$\label{line:updateview}\\
$R.oepoch \gets \max(R.oepoch, oepoch)$\\
$R.status \gets REC$\label{line:setrec}\\
reply $BeginRecoveryAck$\\
\end{algorithm}


\begin{algorithm}
\caption{$Recovery()$ -- per-key Prepare/Accept/Commit}
\label{alg:jetpack_recovery}
\setcounter{AlgoLine}{43}

\underline{Coordinator $R$, after BeginRecovery quorum}\\
\tcc{Phase 1: collect all key-ids in the witness pool}
broadcast $Prepare(R.jepoch, R.oepoch, "pool.key\_ids")$ to all in $R.old\_view$\\
\If{received $f+1$ $PrepareAck(jepoch, value)$}{
    $key\_id\_set \gets \bigcup \{ value \mid jepoch = R.jepoch \}$\label{line:unionkeyids}\\
}
$sid \gets$ fresh global unique set id\\
$rid \gets 0$\\

\BlankLine
\tcc{Phase 2: per-key Prepare then Accept}
\For{each $key$ in $key\_id\_set$\label{line:foreachkey}}{
    broadcast $Prepare(R.jepoch, R.oepoch, key, R.witness.seen\_ballot[key])$ to all in $R.old\_view$\\
    \If{received $f+1$ $PrepareAck(acc\_bal, ins)$\label{line:perkeyprepareack}}{
        $ins\_set \gets$ all $ins$ with largest $jepoch$\\
        \If{$\ge \lceil\frac{f+2}{2}\rceil$ of $ins\_set$ have $cmd$ that preaccepted on fast path\label{line:majority}}{
            $cmd \gets$ that $cmd$\\
            broadcast $Accept(R.jepoch, R.oepoch, sid, rid, cmd)$ to all in $R.old\_view$\label{line:perkeyaccept}\\
            $rid \gets rid + 1$\\
        }
    }
}

\BlankLine
\tcc{Phase 3: Paxos agree on the set descriptor}
$R.Prepare((sid, rid))$\label{line:prepareset}\tcp*{agree on (sid, set-size)}
\end{algorithm}

\paragraph{Key-ID Collection (Phase 1)}
Line~\ref{line:unionkeyids} collects the union of all key-ids seen by the quorum. Because each replica's \texttt{PrepareAck} for the \texttt{"pool.key\_ids"} key returns the set of keys with in-flight fast-path candidates, the union covers every key where a fast-path success \emph{might} have occurred.

\paragraph{Per-Key Decision (Phase 2)}
For each collected key (line~\ref{line:foreachkey}), the coordinator runs a full Paxos Prepare on that key to learn what the quorum holds for it. If at least $\lceil\frac{f+2}{2}\rceil$ replicas preaccepted the same command at that key (line~\ref{line:majority}), that command might have been fast-path-committed, so the coordinator must accept it. This threshold ensures at most one command per key can satisfy this condition (uniqueness of fast-path commands), so the coordinator issues an \texttt{Accept} only for provably fast-path-committed commands.

\paragraph{Set Descriptor Agreement (Phase 3)}
Line~\ref{line:prepareset} runs a Paxos Prepare/Accept/Commit cycle on the pair $(sid, rid)$ — the set identifier and its cardinality. This ensures that even if multiple coordinators race, exactly one $(sid, rid)$ value is committed. After this, Resubmit reads $rec\_set[sid]$ which contains the $rid$ per-key commands accepted in Phase 2.


\begin{algorithm}
\caption{$Prepare(key, default\_value)$ -- per-key Paxos Prepare}
\label{alg:jetpack_prepare}
\setcounter{AlgoLine}{64}

broadcast $Prepare(R.jepoch, R.oepoch, key, R.witness.seen\_ballot[key])$ to all in $R.old\_view$\\
\If{received $f+1$ $PrepareAck(acc\_bal, replied\_value)$}{
    \eIf{highest $acc\_bal > 0$}{
        $to\_acc \gets replied\_value$ with highest $acc\_bal$\\
    }{
        \eIf{$default\_value \ne nil$}{
            $to\_acc \gets default\_value$\\
        }{
            $to\_acc \gets replied\_value$ with highest frequency among responses\\
        }
    }
    $R.Accept(key, to\_acc)$\\
}

\vskip 6pt

\underline{Replica $R$, on receiving $Prepare(jepoch, oepoch, key, ballot)$}\\
\If{$jepoch \ge R.jepoch$ \textbf{and} $oepoch \ge R.oepoch$
    \textbf{and} $ballot \ge R.witness.seen\_ballot[key]$}{
    $R.witness.seen\_ballot[key] \gets ballot$\\
    reply $PrepareAck(R.witness.acc\_ballot[key], R.witness.value[key])$\\
}
\end{algorithm}


\begin{algorithm}
\caption{$Accept(key, value)$ -- per-key Paxos Accept}
\label{alg:jetpack_accept}
\setcounter{AlgoLine}{80}

broadcast $Accept(R.jepoch, R.oepoch, key, R.witness.seen\_ballot[key], value)$ to all in $R.old\_view$\\
\If{received $f+1$ $AcceptAck$}{
    $Commit(key, value)$\\
}

\vskip 6pt

\underline{Replica $R$, on receiving $Accept(jepoch, oepoch, key, ballot, value)$}\\
\If{$jepoch \ge R.jepoch$ \textbf{and} $oepoch \ge R.oepoch$
    \textbf{and} $ballot \ge R.witness.seen\_ballot[key]$
    \textbf{and not} $R.witness.committed[key]$}{
    $R.witness.acc\_ballot[key] \gets ballot$\\
    $R.witness.value[key] \gets value$\\
    reply $AcceptAck$\\
}
\end{algorithm}


\begin{algorithm}
\caption{$Commit(key, value)$ and $Resubmit()$}
\label{alg:jetpack_commit_resubmit}
\setcounter{AlgoLine}{90}

\underline{Coordinator $R$, $Commit(key, value)$}\\
broadcast $Commit(R.jepoch, R.oepoch, key, value)$ to all in $R.old\_view$\\
\tcc{After committing all keys:}
$R.Resubmit()$\\

\vskip 6pt

\underline{Replica $R$, on receiving $Commit(jepoch, oepoch, key, value)$}\\
\If{$jepoch \ge R.jepoch$ \textbf{and} $oepoch \ge R.oepoch$}{
    $R.witness.value[key] \gets value$\\
    $R.witness.committed[key] \gets true$\\
}

\vskip 6pt

\underline{Coordinator $R$, $Resubmit()$}\\
\For{$rid$ from $0$ to $R.rec\_set[R.witness.sid].size - 1$\label{line:resubmitloop}}{
    \If{$R.rec\_set[R.witness.sid][rid] = nil$\label{line:missingcheck}}{
        broadcast $PullRecSetIns(R.jepoch, R.oepoch, R.witness.sid, rid)$\\
        \If{received $PullRecSetInsAck(cmd)$}{
            $R.rec\_set[R.witness.sid][rid] \gets cmd$\\
        }
    }
    $cmd \gets R.rec\_set[R.witness.sid][rid]$\\
    $BroadcastDispatch(R.oepoch, cmd)$\tcp*{resubmit to original protocol}
    wait for $cmd$ result\\
}
broadcast $FinishRecovery(R.oepoch, R.new\_view)$ to all replicas\label{line:finishrecovery}\\
\end{algorithm}

\paragraph{Missing Command Pull}
Line~\ref{line:missingcheck} handles the case where the coordinator has an entry in its recovery set for $(sid, rid)$ but lacks the actual command bytes. This occurs if the coordinator was not among the replicas that witnessed the command. In the non-optimized version, this requires an additional broadcast \texttt{PullRecSetIns} per missing entry. The optimized version eliminates this by bundling the missing-command retrieval into the \texttt{RecordCmd} broadcast in Round 2.


\begin{algorithm}
\caption{Server -- $OnFinishRecovery$}
\label{alg:on_finish_recovery}
\setcounter{AlgoLine}{114}

\underline{Replica $R$, on receiving $FinishRecovery(oepoch, view)$}\\
\If{$oepoch \ge R.oepoch$}{
    $R.jepoch \gets oepoch$\\
    $R.oepoch \gets oepoch$\\
    $R.old\_view \gets view$\\
    $R.new\_view \gets view$\\
    $R.witness.reset()$\tcp*{clear all per-key state}
    $R.status \gets RDY$\tcp*{re-enable fast path}
}
\end{algorithm}


\paragraph{RTT Analysis (Non-Optimized)}
Under a WAN RTT of $\Delta$, for $N$ in-flight fast-path commands at recovery time:
\begin{itemize}
  \setlength{\itemsep}{0pt}
  \item \texttt{BeginRecovery} broadcast: $1 \times \Delta$
  \item $1$ \texttt{Prepare} for pool key-ids: $1 \times \Delta$
  \item $N$ per-key \texttt{Prepare} broadcasts: $N \times \Delta$
  \item $N$ per-key \texttt{Accept} broadcasts: $N \times \Delta$
  \item $N$ per-key \texttt{Commit} broadcasts: $N \times \Delta$ (or fire-and-forget, $0\Delta$)
  \item \texttt{Prepare}(\emph{sid, rid}) + \texttt{Accept} + \texttt{Commit}: $3 \times \Delta$
\end{itemize}
Total recovery latency: $(5 + 3N)\Delta$ in the sequential case. The optimized version reduces this to $2\Delta$ regardless of $N$.

\paragraph{Correctness Notes}
\begin{itemize}
  \setlength{\itemsep}{0pt}
  \item \textbf{Safety}: The per-key Paxos Prepare/Accept/Commit ensures that any command that succeeded on the fast path (witnessed by a $\lceil\frac{f+2}{2}\rceil$ supermajority) is detected and resubmitted. No such command is lost.
  \item \textbf{Uniqueness}: At most one command can satisfy the supermajority threshold per key (Lemma: two commands with $\ge\lceil\frac{f+2}{2}\rceil$ witnesses each would require $\ge 2\lceil\frac{f+2}{2}\rceil > 2f+1$ witnesses total, which is impossible).
  \item \textbf{Liveness}: If the coordinator fails during recovery, the next proposer will run recovery again. The Paxos ballot negotiation prevents two coordinators from accepting conflicting values.
\end{itemize}
